/// 初始化 P2P 模块
///
/// # Arguments
/// * `device_name` - 本设备的显示名称
/// * `work_dir` - 工作目录路径（所有数据将存储在此目录下的子目录中）
///
/// # 目录结构
/// - `{work_dir}/data/` - 数据库文件
/// - `{work_dir}/logs/` - 日志文件
/// - `{work_dir}/certs/` - 证书文件（identity.key）
#[frb(sync)]
pub fn p2p_init(device_name: String, work_dir: String) -> Result<(), String> {
    crate::internal_init(device_name, work_dir)
}
